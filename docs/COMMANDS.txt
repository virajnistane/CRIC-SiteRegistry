🐳 DOCKER MODE (Production-like)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  Start:     ./run-docker.sh
  Stop:      docker compose down  (or Ctrl+C)
  URL:       http://localhost:8000
  Server:    Gunicorn (3 workers)
  Logs:      docker compose logs -f web

💻 LOCAL MODE (Fast development)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  Start:     ./run-local.sh
  Stop:      Ctrl+C
  URL:       http://localhost:8000  
  Server:    Django runserver (auto-reload)
  Shell:     source .venv/bin/activate

🔧 COMMON COMMANDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  Docker:

  Local:
    python manage.py <command>
    pytest

  Migrations:
    Docker:  docker compose exec web python manage.py migrate

  Shell:
